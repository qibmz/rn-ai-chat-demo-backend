- [ ] **Step 1: 更新 `schema.prisma`**

在 `User` 上增加 `bots Bot[]`。

在 `Conversation` 增加：

```prisma
  botId     String    @unique @map("bot_id") @db.Uuid
  bot       Bot       @relation(fields: [botId], references: [id], onDelete: Restrict)
```

（最终态：`botId` NOT NULL + UNIQUE；migration SQL 负责分阶段。）

在文件中追加 enums + `Bot`：

```prisma
enum BotKind {
  main
  sub
}

enum BotFaceShape {
  circle
  oval
  squircle
}

model Bot {
  id           String       @id @default(uuid()) @db.Uuid
  userId       String       @map("user_id") @db.Uuid
  name         String
  kind         BotKind
  faceColor    String       @map("face_color")
  faceShape    BotFaceShape @map("face_shape")
  systemPrompt String?      @map("system_prompt") @db.Text
  deletedAt    DateTime?    @map("deleted_at")
  createdAt    DateTime     @default(now()) @map("created_at")
  updatedAt    DateTime     @updatedAt @map("updated_at")

  user         User          @relation(fields: [userId], references: [id], onDelete: Cascade)
  conversation Conversation?

  @@index([userId, updatedAt])
  @@index([userId, kind])
  @@map("bots")
}
```

部分唯一索引**不要**写在 Prisma schema（无法表达）；只放 migration SQL。

- [ ] **Step 2: 手写 migration SQL**

创建目录 `prisma/migrations/20261013120000_add_bots/`，内容：

```sql
-- CreateEnum
CREATE TYPE "BotKind" AS ENUM ('main', 'sub');
CREATE TYPE "BotFaceShape" AS ENUM ('circle', 'oval', 'squircle');

-- CreateTable bots
CREATE TABLE "bots" (
    "id" UUID NOT NULL,
    "user_id" UUID NOT NULL,
    "name" TEXT NOT NULL,
    "kind" "BotKind" NOT NULL,
    "face_color" TEXT NOT NULL,
    "face_shape" "BotFaceShape" NOT NULL,
    "system_prompt" TEXT,
    "deleted_at" TIMESTAMP(3),
    "created_at" TIMESTAMP(3) NOT NULL DEFAULT CURRENT_TIMESTAMP,
    "updated_at" TIMESTAMP(3) NOT NULL,
    CONSTRAINT "bots_pkey" PRIMARY KEY ("id")
);

CREATE INDEX "bots_user_id_updated_at_idx" ON "bots"("user_id", "updated_at");
CREATE INDEX "bots_user_id_kind_idx" ON "bots"("user_id", "kind");

ALTER TABLE "bots" ADD CONSTRAINT "bots_user_id_fkey"
  FOREIGN KEY ("user_id") REFERENCES "users"("id") ON DELETE CASCADE ON UPDATE CASCADE;

-- 1) 可空 bot_id（尚无 FK / UNIQUE）
ALTER TABLE "conversations" ADD COLUMN "bot_id" UUID;

-- 2) 数据回填：每用户保证有未删 main；未删会话挂 bot；已删会话挂已删 sub
-- 无未删会话的用户也建 main + 空会话
INSERT INTO "bots" ("id", "user_id", "name", "kind", "face_color", "face_shape", "deleted_at", "created_at", "updated_at")
SELECT gen_random_uuid(), u."id", '牛马', 'main', '#0A0A0A', 'circle', NULL, CURRENT_TIMESTAMP, CURRENT_TIMESTAMP
FROM "users" u
WHERE NOT EXISTS (
  SELECT 1 FROM "bots" b WHERE b."user_id" = u."id" AND b."kind" = 'main' AND b."deleted_at" IS NULL
);

-- 每用户第一条未删会话 → 主 Bot
WITH ranked AS (
  SELECT c."id" AS conversation_id, c."user_id",
         ROW_NUMBER() OVER (PARTITION BY c."user_id" ORDER BY c."created_at" ASC, c."id" ASC) AS rn
  FROM "conversations" c
  WHERE c."deleted_at" IS NULL AND c."bot_id" IS NULL
),
mains AS (
  SELECT b."id" AS bot_id, b."user_id"
  FROM "bots" b
  WHERE b."kind" = 'main' AND b."deleted_at" IS NULL
)
UPDATE "conversations" c
SET "bot_id" = m.bot_id
FROM ranked r
JOIN mains m ON m."user_id" = r."user_id"
WHERE c."id" = r.conversation_id AND r.rn = 1;

-- 其余未删会话 → 各建 sub（name = title）
INSERT INTO "bots" ("id", "user_id", "name", "kind", "face_color", "face_shape", "deleted_at", "created_at", "updated_at")
SELECT gen_random_uuid(), c."user_id", c."title", 'sub', '#DB2777', 'circle', NULL, CURRENT_TIMESTAMP, CURRENT_TIMESTAMP
FROM "conversations" c
WHERE c."deleted_at" IS NULL AND c."bot_id" IS NULL;

UPDATE "conversations" c
SET "bot_id" = b."id"
FROM "bots" b
WHERE c."bot_id" IS NULL
  AND c."deleted_at" IS NULL
  AND b."user_id" = c."user_id"
  AND b."kind" = 'sub'
  AND b."deleted_at" IS NULL
  AND b."name" = c."title"
  AND NOT EXISTS (
    SELECT 1 FROM "conversations" c2 WHERE c2."bot_id" = b."id"
  );

-- 上面用 name=title 匹配在同名「新对话」时可能一对多歧义：改用显式临时列更稳。
-- **实现时必须改成**：先为每个无 bot_id 的会话 INSERT bot RETURNING，再 UPDATE。
-- 推荐用 DO $$ 循环或：

-- 已软删会话 → 已删 sub
-- （实现时同样：逐行 INSERT bot(deleted_at=c.deleted_at) 再挂上，禁止靠 name 匹配）

-- 无会话的用户：为 main 建空 Conversation
INSERT INTO "conversations" ("id", "user_id", "title", "bot_id", "created_at", "updated_at")
SELECT gen_random_uuid(), b."user_id", b."name", b."id", CURRENT_TIMESTAMP, CURRENT_TIMESTAMP
FROM "bots" b
WHERE b."kind" = 'main' AND b."deleted_at" IS NULL
  AND NOT EXISTS (SELECT 1 FROM "conversations" c WHERE c."bot_id" = b."id");

-- 3) 断言
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM "conversations" WHERE "bot_id" IS NULL) THEN
    RAISE EXCEPTION 'conversations.bot_id still null after backfill';
  END IF;
END $$;

-- 4–6) NOT NULL + UNIQUE + FK + 部分唯一
ALTER TABLE "conversations" ALTER COLUMN "bot_id" SET NOT NULL;
CREATE UNIQUE INDEX "conversations_bot_id_key" ON "conversations"("bot_id");
ALTER TABLE "conversations" ADD CONSTRAINT "conversations_bot_id_fkey"
  FOREIGN KEY ("bot_id") REFERENCES "bots"("id") ON DELETE RESTRICT ON UPDATE CASCADE;

CREATE UNIQUE INDEX "bots_user_main_undeleted_key"
  ON "bots" ("user_id")
  WHERE "kind" = 'main' AND "deleted_at" IS NULL;
```

**重要：** 上段「靠 name=title 匹配」在同名标题下不安全。实现本 Task 时，回填「其余未删 / 已删会话」**必须**用「逐会话 INSERT Bot 再 UPDATE conversation.bot_id」写法（PL/pgSQL `FOR r IN ... LOOP` 或临时 mapping 表）。上面 SQL 是结构骨架；提交前自检：`SELECT count(*) FROM conversations WHERE bot_id IS NULL` 为 0，且 `bots` 行数 ≥ 用户数。

```ts
// src/chat/bots.service.ts
import {
  BadRequestException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import {
  BotFaceShape,
  BotKind,
  Prisma,
  type Bot,
  type Conversation,
} from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service.js';
import { CreateBotDto } from './dto/create-bot.dto.js';
import { UpdateBotDto } from './dto/update-bot.dto.js';

export type BotDto = {
  id: string;
  name: string;
  kind: 'main' | 'sub';
  faceColor: string;
  faceShape: 'circle' | 'oval' | 'squircle';
  systemPrompt: string | null;
  conversationId: string;
  lastMessagePreview: string | null;
  lastMessageAt: string | null;
  createdAt: string;
  updatedAt: string;
};

const PREVIEW_MAX = 80;
const MAIN_FACE_COLOR = '#0A0A0A';
const MAIN_FACE_SHAPE = BotFaceShape.circle;

type BotWithConversation = Bot & { conversation: Conversation | null };

@Injectable()
export class BotsService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly configService: ConfigService,
  ) {}

  private appNameSnapshot(): string {
    return this.configService.get<string>('APP_NAME', '牛马')?.trim() || '牛马';
  }

  private toDto(
    row: BotWithConversation,
    preview: { text: string | null; at: Date | null },
  ): BotDto {
    if (!row.conversation) {
      throw new Error(`Bot ${row.id} missing conversation`);
    }
    return {
      id: row.id,
      name: row.name,
      kind: row.kind,
      faceColor: row.faceColor,
      faceShape: row.faceShape,
      systemPrompt: row.systemPrompt,
      conversationId: row.conversation.id,
      lastMessagePreview: preview.text,
      lastMessageAt: preview.at ? preview.at.toISOString() : null,
      createdAt: row.createdAt.toISOString(),
      updatedAt: row.updatedAt.toISOString(),
    };
  }

  private formatPreview(content: string, parts: unknown): string | null {
    const trimmed = content?.trim() ?? '';
    if (trimmed) {
      return trimmed.length > PREVIEW_MAX
        ? trimmed.slice(0, PREVIEW_MAX)
        : trimmed;
    }
    const arr = Array.isArray(parts) ? parts : [];
    if (
      arr.some(
        (p) =>
          p &&
          typeof p === 'object' &&
          (p as { type?: string }).type === 'image',
      )
    ) {
      return '[图片]';
    }
    return null;
  }

  /** 一次查出多个 conversation 的最近 user/assistant 消息 */
  private async loadPreviews(
    conversationIds: string[],
  ): Promise<Map<string, { text: string | null; at: Date | null }>> {
    const map = new Map<string, { text: string | null; at: Date | null }>();
    if (conversationIds.length === 0) {
      return map;
    }
    const rows = await this.prisma.$queryRaw<
      Array<{
        conversation_id: string;
        content: string;
        parts: unknown;
        created_at: Date;
      }>
    >(Prisma.sql`
      SELECT DISTINCT ON (m.conversation_id)
        m.conversation_id,
        m.content,
        m.parts,
        m.created_at
      FROM messages m
      WHERE m.conversation_id IN (${Prisma.join(conversationIds)})
        AND m.role IN ('user'::"MessageRole", 'assistant'::"MessageRole")
      ORDER BY m.conversation_id, m.created_at DESC
    `);
    for (const r of rows) {
      map.set(r.conversation_id, {
        text: this.formatPreview(r.content, r.parts),
        at: r.created_at,
      });
    }
    return map;
  }

  async createMainBotWithConversation(
    tx: Prisma.TransactionClient,
    userId: string,
    name: string,
  ): Promise<{ botId: string; conversationId: string }> {
    const bot = await tx.bot.create({
      data: {
        userId,
        name,
        kind: BotKind.main,
        faceColor: MAIN_FACE_COLOR,
        faceShape: MAIN_FACE_SHAPE,
      },
    });
    const conversation = await tx.conversation.create({
      data: {
        userId,
        title: name,
        botId: bot.id,
      },
    });
    return { botId: bot.id, conversationId: conversation.id };
  }

  async ensureMainBot(userId: string): Promise<BotWithConversation> {
    const existing = await this.prisma.bot.findFirst({
      where: { userId, kind: BotKind.main, deletedAt: null },
      include: { conversation: true },
    });
    if (existing?.conversation) {
      return existing;
    }

    const name = this.appNameSnapshot();
    try {
      return await this.prisma.$transaction(async (tx) => {
        const again = await tx.bot.findFirst({
          where: { userId, kind: BotKind.main, deletedAt: null },
          include: { conversation: true },
        });
        if (again?.conversation) {
          return again;
        }
        const { botId } = await this.createMainBotWithConversation(
          tx,
          userId,
          name,
        );
        const row = await tx.bot.findFirstOrThrow({
          where: { id: botId },
          include: { conversation: true },
        });
        return row;
      });
    } catch (err) {
      if (
        err instanceof Prisma.PrismaClientKnownRequestError &&
        err.code === 'P2002'
      ) {
        const row = await this.prisma.bot.findFirst({
          where: { userId, kind: BotKind.main, deletedAt: null },
          include: { conversation: true },
        });
        if (row?.conversation) {
          return row;
        }
      }
      throw err;
    }
  }

  async list(userId: string): Promise<BotDto[]> {
    await this.ensureMainBot(userId);
    const rows = await this.prisma.bot.findMany({
      where: { userId, deletedAt: null },
      include: { conversation: true },
      orderBy: [{ updatedAt: 'desc' }],
    });
    rows.sort((a, b) => {
      if (a.kind === BotKind.main && b.kind !== BotKind.main) return -1;
      if (b.kind === BotKind.main && a.kind !== BotKind.main) return 1;
      return b.updatedAt.getTime() - a.updatedAt.getTime();
    });
    const ids = rows
      .map((r) => r.conversation?.id)
      .filter((id): id is string => !!id);
    const previews = await this.loadPreviews(ids);
    return rows.map((r) =>
      this.toDto(
        r,
        previews.get(r.conversation!.id) ?? { text: null, at: null },
      ),
    );
  }

  async getById(userId: string, id: string): Promise<BotDto> {
    const row = await this.prisma.bot.findFirst({
      where: { id, userId, deletedAt: null },
      include: { conversation: true },
    });
    if (!row?.conversation) {
      throw new NotFoundException('Bot 不存在');
    }
    const previews = await this.loadPreviews([row.conversation.id]);
    return this.toDto(
      row,
      previews.get(row.conversation.id) ?? { text: null, at: null },
    );
  }

  async createSub(userId: string, dto: CreateBotDto): Promise<BotDto> {
    await this.ensureMainBot(userId);
    const row = await this.prisma.$transaction(async (tx) => {
      const bot = await tx.bot.create({
        data: {
          userId,
          name: dto.name.trim(),
          kind: BotKind.sub,
          faceColor: dto.faceColor,
          faceShape: dto.faceShape as BotFaceShape,
          systemPrompt: dto.systemPrompt?.trim() || null,
        },
      });
      await tx.conversation.create({
        data: {
          userId,
          title: bot.name,
          botId: bot.id,
        },
      });
      return tx.bot.findFirstOrThrow({
        where: { id: bot.id },
        include: { conversation: true },
      });
    });
    return this.toDto(row, { text: null, at: null });
  }

  async update(userId: string, id: string, dto: UpdateBotDto): Promise<BotDto> {
    const row = await this.prisma.bot.findFirst({
      where: { id, userId, deletedAt: null },
      include: { conversation: true },
    });
    if (!row?.conversation) {
      throw new NotFoundException('Bot 不存在');
    }

    if (dto.deleted === true) {
      if (row.kind === BotKind.main) {
        throw new BadRequestException('主 Bot 不可删除');
      }
      const updated = await this.prisma.$transaction(async (tx) => {
        const now = new Date();
        await tx.conversation.update({
          where: { id: row.conversation!.id },
          data: { deletedAt: now },
        });
        return tx.bot.update({
          where: { id },
          data: { deletedAt: now },
          include: { conversation: true },
        });
      });
      return this.toDto(updated, { text: null, at: null });
    }

    const data: Prisma.BotUpdateInput = {};
    if (dto.name !== undefined) data.name = dto.name.trim();
    if (dto.faceColor !== undefined) data.faceColor = dto.faceColor;
    if (dto.faceShape !== undefined) {
      data.faceShape = dto.faceShape as BotFaceShape;
    }
    if (dto.systemPrompt !== undefined) {
      data.systemPrompt = dto.systemPrompt.trim() || null;
    }

    const updated = await this.prisma.bot.update({
      where: { id },
      data,
      include: { conversation: true },
    });
    const previews = await this.loadPreviews([updated.conversation!.id]);
    return this.toDto(
      updated,
      previews.get(updated.conversation!.id) ?? { text: null, at: null },
    );
  }
}
```
